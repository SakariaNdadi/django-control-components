schemas

Quickstart

Build responsive form layouts in Python decorating standard Django Form / ModelForm instances.

A Schema decorates an existing Django Form / ModelForm - it never validates itself, Django's form does.

1. Define the Schema

from django_control_components.schemas import Schema, Section, TextInput, Select, Toggle
from myapp.forms import TaskForm

def task_schema():
    return (
        Schema.make()
        .form(TaskForm)
        .schema(
            [
                Section.make("Task Information")
                .columns(2)
                .schema(
                    [
                        TextInput.make("title").required().column_span_full(),
                        Select.make("priority").searchable(),
                        Toggle.make("is_active"),
                    ]
                ),
            ]
        )
    )

2. Wire into Django View

from django.views.generic import CreateView
from django.urls import reverse_lazy
from django_control_components.mixins import SchemaFormMixin
from myapp.models import Task

class TaskCreateView(SchemaFormMixin, CreateView):
    model = Task
    template_name = "tasks/form.html"
    success_url = reverse_lazy("task-list")

    def get_schema(self):
        return task_schema()
