Panels, Resources & Dashboards

An admin-independent surface in pure Python. A Panel mounts a collection of Resource CRUD views and DashboardPage metric grids into your Django routing without touching django.contrib.admin.

1. The Mental Model

A Panel provides a dedicated, branded URL namespace and sidebar navigation. Within a panel:

  • Resource: Maps a single Django Model to four automatic CRUD pages:
    • List Page: Rendered via build_table() with keyset streaming, sorting, search, and bulk actions.
    • Create Page: Rendered via build_schema() validating against Django form classes.
    • Edit Page: Form schema bound to the active model record.
    • View Page: Read-only presentation layout rendered via build_infolist().
  • DashboardPage: Custom analytics/metric dashboards composed of StatWidget, ChartWidget (Chart.js), BarListWidget, and embedded TableWidget.
  • PanelPage: Any custom template-backed page integrated into the panel's navigation structure.

2. Defining a Model Resource

Every method is an explicit classmethod taking request, preventing request cross-contamination:

from django_control_components.panels import Resource
from django_control_components.schemas import Schema, Section, TextInput, Select, Toggle
from django_control_components.tables import Table, TextColumn, BadgeColumn, BooleanColumn, DateColumn, SelectFilter
from django_control_components.infolists import Infolist, TextEntry, BadgeEntry
from myapp.models import Task

class TaskResource(Resource):
    model = Task
    navigation_icon = "list-check"
    navigation_group = "Operations"
    create_redirect = "list"  # "list" or "edit"

    @classmethod
    def can(cls, request, action, obj=None):
        # Fine-grained permissions: "view", "add", "change", "delete"
        return request.user.has_perm(f"myapp.{action}_task")

    @classmethod
    def get_queryset(cls, request):
        # Multi-tenant or row-level scoping
        return cls.model.objects.filter(company=request.user.company)

    @classmethod
    def build_table(cls, *, request):
        return (
            Table.make(cls.get_queryset(request))
            .id("tasks-table")
            .columns([
                TextColumn.make("title").sortable().searchable(),
                BadgeColumn.make("priority").colors({"high": "danger", "medium": "secondary", "low": "muted"}),
                BooleanColumn.make("done").labels(("Done", "Pending")),
                DateColumn.make("due_date").since(),
            ])
            .filters([
                SelectFilter.make("priority").options(Task.Priority.choices),
            ])
            .searchable()
            .default_sort("-due_date")
        )

    @classmethod
    def build_schema(cls, *, request):
        return (
            Schema.make()
            .model(Task, fields=["title", "priority", "done", "due_date"])
            .schema([
                Section.make("Task Details")
                .columns(2)
                .schema([
                    TextInput.make("title").required().column_span_full(),
                    Select.make("priority"),
                    Toggle.make("done"),
                    TextInput.make("due_date"),
                ])
            ])
        )

    @classmethod
    def build_infolist(cls, *, request):
        return (
            Infolist.make().schema([
                TextEntry.make("title"),
                BadgeEntry.make("priority"),
                TextEntry.make("due_date"),
            ])
        )

3. Defining Dashboards & Metric Widgets

from django_control_components.panels import DashboardPage, StatWidget, ChartWidget, BarListWidget
from myapp.models import Task

class AnalyticsDashboard(DashboardPage):
    page_title = "Overview"
    nav_label = "Analytics"
    nav_icon = "chart-pie"

    def widgets(self, request):
        return [
            StatWidget.make("Open Tasks", Task.objects.filter(done=False).count())
                .icon("clock")
                .poll(30),
            ChartWidget.make("Tasks by Priority")
                .kind("bar")
                .data([
                    ("Low", Task.objects.filter(priority="low").count()),
                    ("Medium", Task.objects.filter(priority="medium").count()),
                    ("High", Task.objects.filter(priority="high").count()),
                ])
                .columns(2),
            BarListWidget.make("Completion Breakdown")
                .data([("Complete", 24), ("In Progress", 6)]),
        ]

4. Mounting the Panel in `urls.py`

from django.urls import path
from django_control_components.panels import Panel
from myapp.panels import TaskResource, AnalyticsDashboard

admin_panel = (
    Panel("admin")
    .path("admin-panel")
    .brand("Operations Hub", "shield-halved")
    .resources([TaskResource])
    .pages([AnalyticsDashboard])
    .auth(lambda request: request.user.is_staff)
)

urlpatterns = [
    admin_panel.mount(),  # Routes all /admin-panel/* URLs
]
