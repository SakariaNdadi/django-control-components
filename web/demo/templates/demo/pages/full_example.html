1. Architecture: panels & resources

The entire catalog and demo app are defined in Python source code:

  • Panel: The root routing container and sidebar navigation manager.
  • Resource: Wires one Django model to List (Table), Create/Edit (Schema), and View (Infolist).
  • DashboardPage + Widget: Metric overview pages combining StatWidget, ChartWidget (Chart.js), BarListWidget, and TableWidget.

2. Defining a Complete Model Resource

from django import forms
from django_control_components.panels import Resource
from django_control_components.schemas import Schema, Section, TextInput, Select, Toggle
from django_control_components.tables import Table, TextColumn, BadgeColumn, BooleanColumn, DateColumn, SelectFilter
from django_control_components.infolists import Infolist, TextEntry, BadgeEntry, BooleanEntry, DateEntry
from myapp.models import Task

class TaskResource(Resource):
    model = Task
    navigation_icon = "list-check"
    navigation_group = "Operations"
    create_redirect = "list"

    @classmethod
    def can(cls, request, action, obj=None):
        # Open to all in demo, or use standard Django permissions:
        # return request.user.has_perm(f"myapp.{action}_task")
        return True

    # 1. List Page: Keyset-streamed interactive table
    @classmethod
    def build_table(cls, *, request):
        return (
            Table.make(cls.get_queryset(request))
            .id("tasks-table")
            .columns([
                TextColumn.make("title").sortable().searchable(),
                BadgeColumn.make("priority").colors({
                    "low": "muted",
                    "medium": "secondary",
                    "high": "danger",
                }),
                BooleanColumn.make("done").labels(("✓ Done", "- In Progress")),
                DateColumn.make("due_date").since().sortable(),
            ])
            .filters([
                SelectFilter.make("priority").options(Task.Priority.choices),
            ])
            .default_sort("-due_date")
            .searchable()
        )

    # 2. Create & Edit Pages: Bound Schema Form
    @classmethod
    def build_schema(cls, *, request):
        return (
            Schema.make()
            .model(Task, fields=["title", "priority", "done", "due_date"])
            .schema([
                Section.make("Task Information")
                .columns(2)
                .schema([
                    TextInput.make("title").required().column_span_full(),
                    Select.make("priority"),
                    Toggle.make("done"),
                ]),
                TextInput.make("due_date"),
            ])
        )

    # 3. View Record Page: Read-only Infolist
    @classmethod
    def build_infolist(cls, *, request):
        return (
            Infolist.make()
            .schema([
                TextEntry.make("title").label("Task Title"),
                BadgeEntry.make("priority").colors({"high": "danger"}),
                BooleanEntry.make("done").label("Status Completed"),
                DateEntry.make("due_date").label("Target Date"),
            ])
        )

3. Adding a Dashboard with Chart.js Widgets

Dashboard pages compose metrics and charts with zero client scaffolding - Chart.js loads automatically on-demand via CDN when a ChartWidget is rendered:

from django_control_components.panels import (
    DashboardPage,
    StatWidget,
    ChartWidget,
    BarListWidget,
    TableWidget,
)

class AnalyticsDashboard(DashboardPage):
    page_title = "Executive Overview"
    nav_label = "Analytics"
    nav_icon = "chart-line"

    def widgets(self, request):
        return [
            # Metric Card with auto-polling (every 30s)
            StatWidget.make("Total Tasks", Task.objects.count())
                .icon("list-check")
                .poll(30),

            # Chart.js Interactive Line/Bar/Area/Pie Chart
            ChartWidget.make("Tasks by Priority")
                .kind("bar")
                .data([
                    ("Low Priority", Task.objects.filter(priority="low").count()),
                    ("Medium Priority", Task.objects.filter(priority="medium").count()),
                    ("High Priority", Task.objects.filter(priority="high").count()),
                ])
                .columns(2),

            # Lightweight CSS-only Bar List (Zero JS)
            BarListWidget.make("Breakdown")
                .data([("Done", 12), ("Pending", 4)]),
        ]

4. Mounting the Panel

# app/panels.py
from django_control_components.panels import Panel

app_panel = (
    Panel("admin")
    .path("admin-panel")
    .brand("Enterprise Control", "shield-halved")
    .resources([TaskResource])
    .pages([AnalyticsDashboard])
    .auth(lambda request: request.user.is_staff)
)

# config/urls.py
urlpatterns = [
    app_panel.mount(),  # Serves /admin-panel/, /admin-panel/tasks/, etc.
]
